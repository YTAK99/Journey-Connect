# 1. Vite 도구로 프론트엔드 리액트(JavaScript기반) 프로젝트 생성
npm create vite@latest jc-frontend -- --template react
(which linter to use? -> ESLint)

# 2. 새로 생성된 프론트엔드 폴더로 진입
cd jc-frontend

# 3. 기본 패키지 파일 다운로드
npm install

# 4. 프로젝트 핵심 외부 라이브러리(라우터, HTTP 통신, 전역 상태관리, 아이콘) 한 번에 설치
npm install react-router-dom axios zustand lucide-react

# 5. Tailwind CSS 스타일링 환경 빌드 전용 모듈 설치
npm install -D tailwindcss@3 postcss autoprefixer
npx tailwindcss init -p

# npm ci가 package-lock.json을 기준으로 기존 패키지와 새로 추가된 Tiptap·DOMPurify까지 모두 설치
npm ci
